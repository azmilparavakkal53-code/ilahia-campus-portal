          function findRelevantAIKnowledge(query, entries = aiKnowledgeEntries) {
             const words = query
                 .toLowerCase()
                 .replace(/[^\w\s]/g, ' ')
                 .split(/\s+/)
                 .filter(word => word.length > 2);
             if (!words.length) return null;
             let bestMatch = null;
             let bestScore = 0;
             entries.forEach(entry => {
                 const searchable = [
                     entry.title,
                     entry.content,
                     entry.category,
                     entry.source,
                     ...(entry.tags || [])
                 ].join(' ').toLowerCase();
                 const score = words.reduce((total, word) => total + (searchable.includes(word) ? 1 : 0), 0);
                 if (score > bestScore) {
                     bestScore = score;
                     bestMatch = entry;
                 }
             });
             return bestMatch;
         }

        let translationRun = 0;
        async function translateText(text, targetLang = currentLang, sourceLang = 'en') {
            const source = String(text || '').trim();
            if (!source || targetLang === sourceLang) return source;
            const cacheKey = `${sourceLang}:${targetLang}:${source}`;
            if (translationCache.has(cacheKey)) return translationCache.get(cacheKey);
            if (translationRequests.has(cacheKey)) return translationRequests.get(cacheKey);
            const protectedValues = [];
            let translatableSource = source;
            protectedInstitutionNames.forEach((name, index) => {
                const escapedName = name.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
                translatableSource = translatableSource.replace(new RegExp(escapedName, 'gi'), match => {
                    const token = `ZXQINSTITUTION${index}TOKENQXZ`;
                    protectedValues.push({ token, match });
                    return token;
                });
            });
            const request = (async () => {
            try {
                const url = new URL('https://translate.googleapis.com/translate_a/single');
                url.searchParams.set('client', 'gtx');
                url.searchParams.set('sl', sourceLang);
                url.searchParams.set('tl', translationLanguages[targetLang] || 'en');
                url.searchParams.set('dt', 't');
                url.searchParams.set('q', translatableSource);
                const response = await fetch(url);
                if (!response.ok) throw new Error(`Translation service returned ${response.status}`);
                const data = await response.json();
                let translated = (data[0] || []).map(part => part[0] || '').join('').trim();
                if (!translated) throw new Error('Translation service returned no text');
                protectedValues.forEach(({ token, match }) => {
                    translated = translated.replace(new RegExp(token, 'gi'), match);
                });
                translationCache.set(cacheKey, translated);
                return translated;
            } catch (error) {
                console.warn('Translation unavailable; showing the original text.', error);
                return source;
            }
            })();
            translationRequests.set(cacheKey, request);
            const result = await request;
            translationRequests.delete(cacheKey);
            return result;
        }
        async function runTranslationJobs(jobs, concurrency = 4) {
            let next = 0;
            const workers = Array.from({ length: Math.min(concurrency, jobs.length) }, async () => {
                while (next < jobs.length) {
                    const job = jobs[next++];
                    await job();
                }
            });
            await Promise.all(workers);
        }
        function getTranslatableTextNodes() {
            const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
                acceptNode(node) {
                    const parent = node.parentElement;
                    if (!parent || !node.textContent.trim() || parent.closest('script,style,noscript,code,pre,option')) return NodeFilter.FILTER_REJECT;
                    return NodeFilter.FILTER_ACCEPT;
                }
            });
            const nodes = [];
            while (walker.nextNode()) nodes.push(walker.currentNode);
            return nodes;
        }
        async function translatePage(lang) {
            const run = ++translationRun;
            const nodes = getTranslatableTextNodes();
            const translatableAttributes = ['placeholder', 'title', 'aria-label'];
            if (lang === 'en') {
                nodes.forEach(node => {
                    if (originalTextNodes.has(node)) node.textContent = originalTextNodes.get(node);
                });
                document.querySelectorAll('input,textarea,button,[aria-label],[title]').forEach(element => {
                    const saved = originalAttributes.get(element);
                    if (saved) Object.entries(saved).forEach(([name, value]) => element.setAttribute(name, value));
                });
                return;
            }
            const jobs = nodes.map(node => async () => {
                if (!originalTextNodes.has(node)) originalTextNodes.set(node, node.textContent);
                const original = originalTextNodes.get(node);
                const translated = await translateText(original, lang, 'en');
                if (run === translationRun && currentLang === lang) node.textContent = translated;
            });
            document.querySelectorAll('input,textarea,button,[aria-label],[title]').forEach(element => {
                translatableAttributes.forEach(name => {
                    if (!element.hasAttribute(name)) return;
                    let saved = originalAttributes.get(element);
                    if (!saved) { saved = {}; originalAttributes.set(element, saved); }
                    if (!(name in saved)) saved[name] = element.getAttribute(name);
                    jobs.push(async () => {
                        const value = await translateText(saved[name], lang, 'en');
                        if (run === translationRun && currentLang === lang) element.setAttribute(name, value);
                    });
                });
            });
            await runTranslationJobs(jobs);
            document.documentElement.lang = lang;
            document.documentElement.dir = lang === 'ar' ? 'rtl' : 'ltr';
        }

        function escapeHTML(value) {
            return String(value).replace(/[&<>"']/g, char => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[char]));
        }

        async function loadCampusMapFeatures(campusKey, lat, lon) {
            const requestNumber = ++campusMapRequestNumber;
            const status = document.getElementById('campusMapDataStatus');
            const list = document.getElementById('campusMapFeatureList');
            if (list) list.innerHTML = '';
            const query = `[out:json][timeout:18];(way(around:350,${lat},${lon})[building];nwr(around:350,${lat},${lon})[amenity~"school|college|university|library|classroom|workshop"];);out center geom tags;`;
            const endpoints = [
                'https://overpass-api.de/api/interpreter',
                'https://overpass.kumi.systems/api/interpreter'
            ];
            let result = null;
            for (const endpoint of endpoints) {
                try {
                    const response = await fetch(`${endpoint}?data=${encodeURIComponent(query)}`);
                    if (!response.ok) throw new Error(`Map data request failed (${response.status})`);
                    result = await response.json();
                    break;
                } catch (error) {
                    console.warn('Campus detail source unavailable:', error);
                }
            }
            if (requestNumber !== campusMapRequestNumber || campusKey !== currentActiveKey) return;
            if (!result || !Array.isArray(result.elements)) {
                if (status) status.textContent = 'The map is active, but detailed building data could not be loaded. Try again later or use Google Maps directions.';
                return;
            }

            const uniqueElements = [...new Map(result.elements.map(element => [`${element.type}/${element.id}`, element])).values()];
            const features = [];
            uniqueElements.forEach(element => {
                const tags = element.tags || {};
                if (element.type === 'way' && Array.isArray(element.geometry) && element.geometry.length >= 3) {
                    const ring = element.geometry.map(point => [point.lon, point.lat]);
                    ring.push(ring[0]);
                    features.push({ type: 'Feature', geometry: { type: 'Polygon', coordinates: [ring] }, properties: tags });
                } else {
                    const center = element.lat && element.lon ? element : element.center;
                    if (center) features.push({ type: 'Feature', geometry: { type: 'Point', coordinates: [center.lon, center.lat] }, properties: tags });
                }
            });
            campusBuildingsLayer.clearLayers();
            campusBuildingsLayer.addData({ type: 'FeatureCollection', features });
            const namedFeatures = uniqueElements.filter(element => {
                const tags = element.tags || {};
                return tags.name || tags['name:en'] || (tags.building && tags.building !== 'yes');
            }).slice(0, 24);
            if (list) {
                list.innerHTML = namedFeatures.length
                    ? namedFeatures.map(element => {
                        const tags = element.tags || {};
                        const label = tags.name || tags['name:en'] || `${tags.building} building`;
                        return `<li><i class="fa-solid fa-location-dot text-blue-600 mr-1"></i>${escapeHTML(label)}</li>`;
                    }).join('')
                    : '<li>No named buildings are mapped here yet; zoom in to inspect available building footprints.</li>';
            }
            if (status) status.textContent = `${features.length} mapped building or campus feature(s) loaded near this campus from OpenStreetMap. Locations are community-mapped and may be incomplete.`;
        }

             } catch (error) {
                 const localEntry = { id: `local-${Date.now()}`, ...payload };
                 aiKnowledgeEntries.unshift(localEntry);
                 saveAIKnowledgeEntries();
                 updateAdminStats();
                 feedback.className = 'mt-3 p-3 rounded-lg text-xs bg-amber-50 border border-amber-200 text-amber-800';
                 feedback.textContent = 'Database unavailable. Saved locally in this browser only; reconnect before relying on it.';

             document.getElementById('aiKnowledgeTags').value = '';

         }
          async function deleteAIKnowledgeEntry(entryId) {
              const entry = [...aiKnowledgeEntries, ...studentAIKnowledgeEntries].find(item => String(item.id) === String(entryId));
             if (!entry) return;
             if (!confirm(`Delete "${entry.title}" from the AI knowledge database? The assistant will stop using this information.`)) {
                 return;
             }
             try {
                 if (aiKnowledgeDatabaseReady && Number.isFinite(Number(entryId))) {
                     await requestAIKnowledge(`/api/ai-knowledge/${entryId}`, { method: 'DELETE' });
                     await loadAIKnowledgeFromDatabase();
                 } else {
                     aiKnowledgeEntries = aiKnowledgeEntries.filter(item => String(item.id) !== String(entryId));
                     studentAIKnowledgeEntries = studentAIKnowledgeEntries.filter(item => String(item.id) !== String(entryId));
                     saveAIKnowledgeEntries();
                     updateAdminStats();
                 }
                 logSystemActivity(`[AI KNOWLEDGE] Deleted "${entry.title}" from the shared assistant context.`);
             } catch (error) {
                 alert('The database could not delete this entry. Please try again while connected.');
             }
          }

          async function addTeacherAIKnowledge() {
              const title = document.getElementById('teacherKnowledgeTitle').value.trim();
              const category = document.getElementById('teacherKnowledgeCategory').value;
              const tags = document.getElementById('teacherKnowledgeTags').value
                  .split(',')
                  .map(tag => tag.trim())
                  .filter(Boolean);
              const content = document.getElementById('teacherKnowledgeContent').value.trim();
              const pdfInput = document.getElementById('teacherKnowledgePdf');
              const pdf = pdfInput?.files?.[0];
              const feedback = document.getElementById('teacherKnowledgeFeedback');
              if (!title || (content.length < 10 && !pdf)) {
                  feedback.className = 'mt-3 p-3 rounded-lg text-xs bg-red-50 border border-red-200 text-red-800';
                  feedback.textContent = 'Please provide a title and notes, or attach a PDF handout.';
                  return;
              }
              if (pdf && (pdf.type !== 'application/pdf' || pdf.size > 25 * 1024 * 1024)) {
                  feedback.className = 'mt-3 p-3 rounded-lg text-xs bg-red-50 border border-red-200 text-red-800';
                  feedback.textContent = 'Please choose a valid PDF file smaller than 25 MB.';
                  return;
              }
              try {
                  feedback.className = 'mt-3 p-3 rounded-lg text-xs bg-cyan-50 border border-cyan-200 text-cyan-800';
                  feedback.textContent = pdf ? 'Uploading PDF and reading its text...' : 'Saving note...';
                  let attachmentPath = null;
                  let extractedPdfText = '';
                  if (pdf) {
                      const upload = await requestAIKnowledge('/api/storage/uploads/request-url', {
                          method: 'POST',
                          body: JSON.stringify({
                              name: pdf.name,
                              size: pdf.size,
                              contentType: 'application/pdf'
                          })
                      });
                      const uploadResponse = await fetch(upload.uploadURL, {
                          method: 'PUT',
                          headers: { 'Content-Type': 'application/pdf' },
                          body: pdf
                      });
                      if (!uploadResponse.ok) throw new Error('PDF upload failed.');
                      const uploadResult = await uploadResponse.json();
                      attachmentPath = uploadResult.objectPath || upload.objectPath;
                      extractedPdfText = uploadResult.extractedText || '';
                  }
                  const knowledgeContent = [content, extractedPdfText].filter(Boolean).join('\n\n');
                  if (knowledgeContent.trim().length < 10) {
                      throw new Error('No readable text was found in this PDF. Paste the notes into the text box, then upload the PDF again.');
                  }
                  await requestAIKnowledge('/api/ai-knowledge', {
                      method: 'POST',
                      body: JSON.stringify({
                          title,
                          category,
                          audience: 'students',
                          source: pdf ? 'Faculty PDF handout' : 'Faculty note',
                          tags,
                          content: knowledgeContent,
                          attachmentPath,
                          attachmentName: pdf?.name,
                          attachmentSize: pdf?.size
                      })
                  });
                  await loadAIKnowledgeFromDatabase();
                  await loadTeacherKnowledge();
                  document.getElementById('teacherKnowledgeTitle').value = '';
                  document.getElementById('teacherKnowledgeTags').value = '';
                  document.getElementById('teacherKnowledgeContent').value = '';
                  if (pdfInput) pdfInput.value = '';
                  feedback.className = 'mt-3 p-3 rounded-lg text-xs bg-emerald-50 border border-emerald-200 text-emerald-800';
                  feedback.textContent = pdf ? 'PDF saved. Readable PDF text is now available to students in ClassBot.' : 'Note shared. Students can ask ClassBot about it now.';
                  logSystemActivity(`[AI KNOWLEDGE] Faculty ${pdf ? 'PDF handout' : 'note'} "${title}" shared with student ClassBot.`);
              } catch (error) {
                  feedback.className = 'mt-3 p-3 rounded-lg text-xs bg-red-50 border border-red-200 text-red-800';
                  feedback.textContent = error.message || 'The note could not be saved to the shared database. Please try again.';
              }
          }

        async function appendChatReply(inputText, messages, styleClass, entries, fallback, rules) {
            const englishText = currentLang === 'en' ? inputText : await translateText(inputText, 'en', currentLang);
            const lower = englishText.toLowerCase();
            const match = findRelevantAIKnowledge(englishText, entries);
            let reply = match ? match.content : fallback;
            if (!match) {
                for (const [terms, response] of rules) {
                    if (terms.some(term => lower.includes(term))) { reply = response; break; }
                }
            }
            const bubble = document.createElement('div');
            bubble.className = styleClass;
            bubble.textContent = currentLang === 'en' ? reply : await translateText(reply, currentLang, 'en');
            messages.appendChild(bubble);
            messages.scrollTop = messages.scrollHeight;
        }
        async function sendMessage() {
            const input = document.getElementById('chatInput');
            const text = input.value.trim();
            if (!text) return;
            const messages = document.getElementById('chatMessages');
            const bubble = document.createElement('div');
            bubble.className = 'bg-blue-700 text-white p-3 rounded-xl rounded-tr-none ml-auto max-w-[85%] text-xs sm:text-sm';
            bubble.textContent = text;
            messages.appendChild(bubble);
            input.value = '';
            messages.scrollTop = messages.scrollHeight;
            await loadAIKnowledgeFromDatabase();
            await appendChatReply(text, messages, 'bg-blue-50 text-blue-950 p-3 rounded-xl rounded-tl-none max-w-[85%] border border-blue-100 text-xs sm:text-sm', aiKnowledgeEntries,
                'Thank you for your inquiry about Ilahia Trust Institutions. Explore the programs or sign into your portal for details.', [
                    [['fee', 'tuition'], 'Tuition varies by institution and program. Contact the college office for the current fee schedule.'],
                    [['hostel', 'room'], 'Please contact the selected campus for current hostel availability and facilities.'],
                    [['bus', 'transport', 'route'], 'Contact the selected campus for current bus routes and schedules.'],
                    [['admission', 'seat'], 'Contact the admissions office for current eligibility, seats, and application dates.']
                ]);
        }
